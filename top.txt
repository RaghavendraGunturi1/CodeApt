@login_required
def start_exam(request, topic_id):
    topic = get_object_or_404(Topic, id=topic_id)
    try:
        exam = topic.exam
    except Exam.DoesNotExist:
        return render(request, 'error.html', {'message': 'No exam found for this topic.'})
